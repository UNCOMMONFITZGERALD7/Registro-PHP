try {
    $pdo = new PDO($dsn, $user, $password); -- para importar try
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION); -- para setear formato de errores
    echo 'Conexion exitosa'; -- flag

} catch (PDOException $e) { -- catch para error
    echo 'No se entabló la conextión: ' . $e->getMessage(); -- formato de error
}
